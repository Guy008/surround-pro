#!/bin/bash

# =================================================================================
# "התוכנה" - סקריפט מתקדם להמרת וידאו או שמע לצליל היקפי 7.1
#
# גירסה סופית ומתוקנת:
# 1. מותאמת ל-Arch Linux (UV + TorchCodec).
# 2. תמיכה מלאה בשמות קבצים בעברית.
# 3. מנגנון חכם למניעת דריסת קבצים קיימים.
# 4. שמירה על כל הלוגיקה המקורית (פילטרים, טיפול בתיקיות).
# =================================================================================

# --- הגדרות ---
set -e # יציאה מיידית במקרה של שגיאה
SCRIPT_DIR=$(dirname "$(readlink -f "$0")")
cd "$SCRIPT_DIR" || exit

# --- הגדרת דפדפן ל-cookies (לעקיפת בדיקת בוטים של YouTube) ---
# ניתן לשנות ל: firefox, chrome, brave, chromium, edge, opera, vivaldi, safari
BROWSER_FOR_COOKIES="chrome"

# --- פונקציות עזר ---
print_error() { echo -e "\e[31mשגיאה: $1\e[0m"; }
print_info() { echo -e "\e[36mמידע: $1\e[0m"; }
print_success() { echo -e "\e[32mהצלחה: $1\e[0m"; }

# פונקציה למציאת שם קובץ פנוי (למניעת דריסה)
get_unique_filename() {
    local base_path="$1" # נתיב מלא ללא סיומת
    local extension="$2" # הסיומת (למשל .mkv)
    local final_path="${base_path}${extension}"
    local counter=1

    # כל עוד הקובץ קיים, הוסף לו מספר
    while [ -f "$final_path" ]; do
        final_path="${base_path}_${counter}${extension}"
        counter=$((counter + 1))
    done
    echo "$final_path"
}

# --- בדיקת תלויות ---
print_info "בדיקת כלים נדרשים..."
command -v yt-dlp >/dev/null 2>&1 || { print_error "הכלי yt-dlp אינו מותקן. יש להתקין אותו."; exit 1; }
command -v ffmpeg >/dev/null 2>&1 || { print_error "הכלי ffmpeg אינו מותקן. יש להתקין אותו."; exit 1; }
command -v ffprobe >/dev/null 2>&1 || { print_error "הכלי ffprobe אינו מותקן. יש להתקין אותו."; exit 1; }
command -v uv >/dev/null 2>&1 || { print_error "הכלי uv אינו מותקן. התקן אותו באמצעות: sudo pacman -S uv"; exit 1; }

print_success "כל הכלים הנדרשים קיימים."
echo ""

# =================================================================================
# פונקציית עיבוד עבור מקור בודד (קובץ או URL)
# =================================================================================
process_source() {
    local SOURCE_INPUT="$1"
    local IS_URL=false
    local IS_VIDEO=false
    local SOURCE_TITLE=""
    local SAFE_TITLE=""
    local WORK_DIR=""
    local ORIGINAL_MEDIA_FILE="original_media"
    local ORIGINAL_AUDIO_FILE="original_audio.wav"
    local VIDEO_FILE="video.mp4"

    # --- הכנת סביבת עבודה וזיהוי המקור ---
    if [[ "$SOURCE_INPUT" == http* ]]; then
        IS_URL=true
        print_info "זוהתה כתובת URL. מאמת פרטים..."
        SOURCE_TITLE=$(yt-dlp --cookies-from-browser "$BROWSER_FOR_COOKIES" --get-title "$SOURCE_INPUT")
        if yt-dlp --cookies-from-browser "$BROWSER_FOR_COOKIES" --list-formats "$SOURCE_INPUT" | grep -q 'video only'; then
            IS_VIDEO=true
            print_info "המקור הוא וידאו."
        else
            print_info "המקור הוא שמע בלבד."
        fi
    else
        if [ ! -f "$SOURCE_INPUT" ]; then
            print_error "הקובץ המקומי '$SOURCE_INPUT' אינו קיים או אינו קובץ."
            return 1
        fi
        print_info "זוהה קובץ מקומי."
        # חילוץ שם הקובץ ללא סיומת
        SOURCE_TITLE=$(basename "$SOURCE_INPUT" | sed 's/\.[^.]*$//')
        
        if ffprobe -v error -select_streams v:0 -count_packets -show_entries stream=nb_read_packets -of csv=p=0 "$SOURCE_INPUT" | grep -q '[1-9]'; then
            IS_VIDEO=true
            print_info "המקור הוא וידאו."
        else
            print_info "המקור הוא שמע בלבד."
        fi
    fi

    # === תיקון 1: תמיכה בעברית ===
    # במקום למחוק הכל, מחליפים רק תווים שאסורים בשמות קבצים (/ \ : * ? " < > |)
    SAFE_TITLE=$(echo "$SOURCE_TITLE" | sed 's/[/\\?%*:|"<>]/_/g' | xargs)
    if [ -z "$SAFE_TITLE" ]; then SAFE_TITLE="output_processed"; fi

    # שימוש ב-RANDOM כדי למנוע התנגשויות בתיקיות זמניות
    WORK_DIR="${SAFE_TITLE}_7.1_Processing_${RANDOM}"

    rm -rf "$WORK_DIR"
    mkdir -p -- "$WORK_DIR"
    
    pushd "$WORK_DIR" > /dev/null

    # --- שלב 1: הורדה/העתקה והפרדת מקורות ---
    print_info "שלב 1/5: מכין את קובץ המקור..."
    if [ "$IS_URL" = true ]; then
        if [ "$IS_VIDEO" = true ]; then
            print_info "מוריד וידאו ושמע..."
            yt-dlp --cookies-from-browser "$BROWSER_FOR_COOKIES" -f "bestvideo[ext=mp4]+bestaudio[ext=m4a]/best[ext=mp4]" --output "$VIDEO_FILE" "$SOURCE_INPUT"
            ffmpeg -i "$VIDEO_FILE" -vn -acodec pcm_s16le -ar 48000 -ac 2 "$ORIGINAL_AUDIO_FILE"
        else
            print_info "מוריד שמע בלבד..."
            yt-dlp --cookies-from-browser "$BROWSER_FOR_COOKIES" -f bestaudio --extract-audio --audio-format wav --output "$ORIGINAL_AUDIO_FILE" "$SOURCE_INPUT"
        fi
    else # קובץ מקומי
        cp "$SOURCE_INPUT" "$ORIGINAL_MEDIA_FILE"
        if [ "$IS_VIDEO" = true ]; then
            print_info "מפריד וידאו ושמע מקובץ מקומי..."
            ffmpeg -i "$ORIGINAL_MEDIA_FILE" -vcodec copy -an "$VIDEO_FILE"
            ffmpeg -i "$ORIGINAL_MEDIA_FILE" -vn -acodec pcm_s16le -ar 48000 -ac 2 "$ORIGINAL_AUDIO_FILE"
        else
            print_info "מעבד קובץ שמע מקומי..."
            ffmpeg -i "$ORIGINAL_MEDIA_FILE" -acodec pcm_s16le -ar 48000 -ac 2 "$ORIGINAL_AUDIO_FILE"
        fi
    fi
    print_success "הכנת המקור הושלמה."
    echo ""

    # --- שלב 2: הפרדת ערוצים עם Demucs (באמצעות UV) ---
    print_info "שלב 2/5: מפריד את השמע לערוצים עם Demucs (זה עלול לקחת זמן)..."
    DEMUCS_MODEL="htdemucs"
    
    # הרצת Demucs דרך UV
    uv run --python 3.11 --with demucs --with torchcodec --with soundfile demucs -n "$DEMUCS_MODEL" -o . --filename "{stem}.{ext}" "$ORIGINAL_AUDIO_FILE"

    STEMS_DIR="$DEMUCS_MODEL"
    DRUMS_FILE="$STEMS_DIR/drums.wav"
    VOCALS_FILE="$STEMS_DIR/vocals.wav"
    BASS_FILE="$STEMS_DIR/bass.wav"
    OTHER_FILE="$STEMS_DIR/other.wav"

    for f in "$DRUMS_FILE" "$VOCALS_FILE" "$BASS_FILE" "$OTHER_FILE"; do
        if [ ! -f "$f" ]; then
            print_error "קובץ ערוץ חסר לאחר הפרדת Demucs: $f"
            popd > /dev/null; rm -rf "$WORK_DIR"; return 1;
        fi
    done
    print_success "ההפרדה לערוצים הסתיימה."
    echo ""

    # --- שלב 3: מיפוי ואיחוד ערוצים 7.1 עם FFmpeg ---
    print_info "שלב 3/5: ממפה ומאחד את הערוצים לקובץ 7.1..."
    PROCESSED_AUDIO_FILE="final_7.1.flac"

    # הפילטר המקורי נשמר במדויק
    ffmpeg \
        -i "$DRUMS_FILE" \
        -i "$VOCALS_FILE" \
        -i "$BASS_FILE" \
        -i "$OTHER_FILE" \
        -filter_complex " \
            [0:a]asplit=2[drums_hp_in][drums_lp_in]; \
            [drums_hp_in]highpass=f=250,channelsplit=channel_layout=stereo[drums_l][drums_r]; \
            [drums_lp_in]lowpass=f=120,pan=mono|c0=0.5*c0+0.5*c1[drums_bass_mono]; \
            [2:a]pan=mono|c0=0.5*c0+0.5*c1[bass_mono]; \
            [drums_bass_mono][bass_mono]amix=inputs=2[lfe]; \
            [1:a]pan=mono|c0=0.5*c0+0.5*c1[vocals_mono]; \
            [3:a]channelsplit=channel_layout=stereo[other_l][other_r]; \
            [drums_l][drums_r][vocals_mono][lfe][other_l][other_r][other_l][other_r]join=inputs=8:channel_layout=7.1:map=0.0-FL|1.0-FR|2.0-FC|3.0-LFE|4.0-SL|5.0-SR|6.0-BL|7.0-BR[out] \
        " \
        -map "[out]" \
        -c:a flac -ar 48000 -sample_fmt s16 \
        "$PROCESSED_AUDIO_FILE"

    print_success "האיחוד והייצוא ל-FLAC הושלמו."
    echo ""

    # --- שלב 4: איחוד סופי וניקיון ---
    # יציאה מתיקיית העבודה כדי לבדוק שמות קבצים בתיקייה הראשית
    popd > /dev/null

    local FINAL_EXT
    if [ "$IS_VIDEO" = true ]; then
        FINAL_EXT=".mkv"
    else
        FINAL_EXT=".flac"
    fi

    # === תיקון 2: יצירת שם קובץ ייחודי (למניעת דריסה) ===
    # ../${SAFE_TITLE}_7.1  --> הנתיב בתיקייה המקורית
    local TARGET_BASE_NAME="../${SAFE_TITLE}_7.1"
    local FINAL_OUTPUT_FILE
    FINAL_OUTPUT_FILE=$(get_unique_filename "$TARGET_BASE_NAME" "$FINAL_EXT")
    
    # חזרה לתיקיית העבודה לביצוע הפעולות האחרונות
    pushd "$WORK_DIR" > /dev/null

    if [ "$IS_VIDEO" = true ]; then
        print_info "שלב 4/5: מאחד וידאו, שמע 7.1 ו-Metadata לקובץ MKV..."
        local METADATA_SOURCE="$VIDEO_FILE"
        if [ ! "$IS_URL" = true ]; then
            METADATA_SOURCE="$SOURCE_INPUT"
        fi
        # כאן אנחנו משתמשים בנתיב היחסי שחושב קודם (הכולל ../)
        ffmpeg -i "$VIDEO_FILE" -i "$PROCESSED_AUDIO_FILE" -i "$METADATA_SOURCE" -map 0:v:0 -map 1:a:0 -map_metadata 2 -c:v copy -c:a copy "$FINAL_OUTPUT_FILE"
        print_success "קובץ MKV סופי נוצר."
    else
        print_info "שלב 4/5: מייצא קובץ שמע סופי..."
        mv "$PROCESSED_AUDIO_FILE" "$FINAL_OUTPUT_FILE"
        print_success "קובץ FLAC סופי נוצר."
    fi
    echo ""

    # --- שלב 5: ניקיון ---
    print_info "שלב 5/5: מנקה קבצים זמניים..."
    popd > /dev/null
    rm -r "$WORK_DIR"
    print_success "הניקיון הסתיים."
    
    local full_output_path
    full_output_path=$(readlink -f "$FINAL_OUTPUT_FILE")
    print_info "הקובץ הסופי נשמר ב: $full_output_path"
}

# =================================================================================
# לוגיקת הרצה ראשית
# =================================================================================

# --- שלב 1: קבלת קלט מהמשתמש ---
read -p "הזן את כתובת ה-URL, נתיב לקובץ, או נתיב לתיקיה: " USER_INPUT
if [ -z "$USER_INPUT" ]; then
    print_error "לא הוכנס קלט."
    exit 1
fi

# --- בדיקת סוג הקלט והרצת העיבוד ---
if [ -d "$USER_INPUT" ]; then
    ABS_DIR_PATH=$(readlink -f -- "$USER_INPUT")
    print_info "זוהתה תיקיה. מעבד את כל הקבצים בתוכה..."
    
    # שימוש ב-find כדי למצוא קבצים באופן בטוח (גם עם שמות מיוחדים)
    file_count=$(find "$ABS_DIR_PATH" -maxdepth 1 -type f | wc -l)
    if [ "$file_count" -eq 0 ]; then
        print_error "לא נמצאו קבצים בתיקיה '$ABS_DIR_PATH'."
        exit 1
    fi
    
    processed_count=0
    while IFS= read -r -d $'' file; do
        processed_count=$((processed_count + 1))
        echo ""
        print_info "=================================================="
        print_info "מתחיל עיבוד קובץ $processed_count מתוך $file_count: $(basename "$file")"
        print_info "=================================================="
        process_source "$file"
        print_success "העיבוד עבור $(basename "$file") הושלם."
    done < <(find "$ABS_DIR_PATH" -maxdepth 1 -type f -print0)

    echo ""
    print_success "✅ העיבוד של כל $file_count הקבצים בתיקיה הושלם."

elif [[ "$USER_INPUT" == http* ]]; then
    process_source "$USER_INPUT"
    echo ""
    print_success "✅ התהליך כולו הושלם!"

elif [ -f "$USER_INPUT" ]; then
    ABS_FILE_PATH=$(readlink -f -- "$USER_INPUT")
    process_source "$ABS_FILE_PATH"
    echo ""
    print_success "✅ התהליך כולו הושלם!"
else
    print_error "הקלט '$USER_INPUT' אינו קובץ, תיקיה או URL חוקיים."
    exit 1
fi

* מפת v0.1:
    FR + FL   = drums (ללא שינוי)
    FC        = vocals
    LFE       = bass (ללא שינוי)
    SL + SR   = drums (ללא שינוי) + backing vocals (אם נצליח להפריד)
    BL + BR   = all other sounds


-----------------------------------
2. ההבדלים בין הסקריפט הדוגמה למפת v0.1
-----------------------------------

הסקריפט הדוגמה היה מקובץ ל-v3 קודמת, ויש שינויים מהותיים במפה:

| ערוץ      | בסקריפט הדוגמה              | ב-v0.1 לפי גיא           |
|-----------|------------------------------|--------------------------|
| FL / FR   | drums highpass >250Hz       | drums מלאים, ללא סינון  |
| FC        | vocals mono                  | vocals (כנראה mono?)    |
| LFE       | drums lowpass + bass mixed   | bass בלבד               |
| SL / SR   | drums highpass (split L/R)   | drums מלאים + backing   |
| BL / BR   | other (split L/R) — duplicate| "all other sounds"      |

המשמעות: עיקר הפילטרים של ffmpeg משתנה. הצרת הסקריפט תהיה הרבה יותר פשוטה
(כי אין highpass/lowpass), אבל יש לחשוב על כמה דברים — ראה שאלות.


-----------------------------------
3. שאלות הבהרה לפני קוד
-----------------------------------

[A] כפילות drums ב-FL/FR וגם ב-SL/SR
    המפה אומרת: drums מלאים גם בקדמה (FL/FR) וגם בצדדים (SL/SR).
    זה אומר ש"אותו אודיו תופים" מתפזר בארבעה ערוצים — מה שייצור "שדה תופים
    רחב" בחדר. שתי שאלות:
    1. זאת הכוונה? (כן/לא)
    2. אם כן, האם השמאל של ה-drums.wav הולך גם ל-FL וגם ל-SL?
       (זאת אומרת stereo identical) — או שאתה רוצה איזשהו פיצול אקטיבי
       בין הקדמה לצדדים?

[B] backing vocals — האם להפריד או לוותר ב-v0.1?
    המפה אומרת "backing vocals אם נצליח להפריד אותם".
    שתי אפשרויות:
    (1) להוסיף שלב נוסף עם audio-separator + MelBand Karaoke
        (קיים בקוד הישן של surround_71_v3.sh — עובד טוב).
        זה מוסיף ~30 שניות לעיבוד, ועוד תלות פייתון.
    (2) לדלג ב-v0.1 — SL/SR יקבלו רק drums, ה-backing נכנס ל-BL/BR
        ביחד עם other (כלומר נופל בערוצי "all other sounds").

    איזה מהאופציות? (אני נוטה ל-1 כי הקוד כבר קיים בגרסה הישנה,
    אבל זה הוספה שמסבכת — תחליט.)

[C] "All other sounds" ב-BL/BR
    אני מבין: זה ה-stem "other.wav" של Demucs (כל מה שלא vocals/drums/bass).
    כולל גיטרות, סינתים, מקלדות, מיתרים, כלי נשיפה — הכל יחד.
    סטריאו מלא L/R, או mono? (בסקריפט הדוגמה היה stereo split.)

[D] FC — vocals stereo או mono?
    הסקריפט הישן עשה pan=mono (מערבב L+R לתעלה אחת). זה הסטנדרט ל-FC
    בקולנוע. נשאר כך?

    תקבל artifact מוזר בתוצאה הסופית.

3. תיקון: Fallback אוטומטי ל-CPU
-----------------------------------

הוספתי ל-`surround-pro.sh`:

    if ! prefetch_demucs_model "$backend"; then
        if [ "$backend" != "cpu" ]; then
            print_warn "טעינת torch עם backend '$backend' נכשלה..."
            backend="cpu"
            if ! prefetch_demucs_model "$backend"; then
                exit 1
            fi
        fi
